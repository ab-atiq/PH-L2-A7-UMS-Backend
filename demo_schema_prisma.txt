model User {
  id            String     @id @default(uuid())
  email         String     @unique
  passwordHash  String?
  role          Role
  status        UserStatus @default(PENDING_VERIFICATION)
  firstName     String
  lastName      String
  phone         String?
  avatarUrl     String?
  googleId      String?    @unique
  emailVerified Boolean    @default(false)
  lastLoginAt   DateTime?

  studentProfile StudentProfile?
  facultyProfile FacultyProfile?

  refreshTokens RefreshToken[]
  auditLogs     AuditLog[]
  notifications Notification[]

  createdAt DateTime  @default(now())
  updatedAt DateTime  @updatedAt
  deletedAt DateTime?

  @@index([email])
  @@index([role])
  @@index([status])
  @@index([createdAt])
  @@map("users")
}

model RefreshToken {
  id        String    @id @default(uuid())
  userId    String
  user      User      @relation(fields: [userId], references: [id])
  tokenHash String    @unique
  expiresAt DateTime
  revoked   Boolean   @default(false)
  revokedAt DateTime?
  userAgent String?
  ipAddress String?

  createdAt DateTime @default(now())

  @@index([userId])
  @@index([expiresAt])
  @@map("refresh_tokens")
}

model StudentProfile {
  id                String      @id @default(uuid())
  userId            String      @unique
  user              User        @relation(fields: [userId], references: [id])
  studentId         String      @unique
  programId         String?
  program           Program?    @relation(fields: [programId], references: [id])
  departmentId      String?
  department        Department? @relation(fields: [departmentId], references: [id])
  currentSemesterId String?
  currentSemester   Semester?   @relation(fields: [currentSemesterId], references: [id])

  batchYear     Int?
  gender        Gender?
  dateOfBirth   DateTime?
  address       String?
  guardianName  String?
  guardianPhone String?
  admissionDate DateTime?

  enrollments Enrollment[]
  attendances Attendance[]
  results     Result[]
  feeInvoices FeeInvoice[]
  payments    Payment[]

  createdAt DateTime  @default(now())
  updatedAt DateTime  @updatedAt
  deletedAt DateTime?

  @@index([studentId])
  @@index([programId])
  @@index([departmentId])
  @@map("student_profiles")
}

model FacultyProfile {

  @@index([departmentId])

}

model Department {
  id          String       @id @default(uuid())
  name        String
  code        String       @unique
  description String?
  status      EntityStatus @default(ACTIVE)

  programs        Program[]
  courses         Course[]
  studentProfiles StudentProfile[]
  facultyProfiles FacultyProfile[]

  createdAt DateTime  @default(now())
  updatedAt DateTime  @updatedAt
  deletedAt DateTime?

  @@index([code])
  @@index([status])
  @@map("departments")
}

model Program {
  id            String       @id @default(uuid())
  name          String
  code          String       @unique
  departmentId  String
  department    Department   @relation(fields: [departmentId], references: [id])
  durationYears Int
  totalCredits  Int
  status        EntityStatus @default(ACTIVE)

  students StudentProfile[]

  createdAt DateTime  @default(now())
  updatedAt DateTime  @updatedAt
  deletedAt DateTime?

  @@index([code])
  @@index([departmentId])
  @@index([status])
  @@map("programs")
}

model Course {
  id           String       @id @default(uuid())
  courseCode   String       @unique
  title        String
  description  String?
  credits      Int
  departmentId String
  department   Department   @relation(fields: [departmentId], references: [id])
  status       CourseStatus @default(DRAFT)

  prerequisitesFor CoursePrerequisite[] @relation("CourseToPrereq")
  prerequisiteOf   CoursePrerequisite[] @relation("PrereqToCourse")
  sections         Section[]

  createdAt DateTime  @default(now())
  updatedAt DateTime  @updatedAt
  deletedAt DateTime?

  @@index([courseCode])
  @@index([departmentId])
  @@index([status])
  @@map("courses")
}

// Composite unique so a course cannot list the same prerequisite twice
model CoursePrerequisite {
  id             String @id @default(uuid())
  courseId       String
  course         Course @relation("CourseToPrereq", fields: [courseId], references: [id])
  prerequisiteId String
  prerequisite   Course @relation("PrereqToCourse", fields: [prerequisiteId], references: [id])

  createdAt DateTime @default(now())

  @@unique([courseId, prerequisiteId])
  @@index([courseId])
  @@index([prerequisiteId])
  @@map("course_prerequisites")
}

model Semester {
  id                String         @id @default(uuid())
  name              String         @unique
  startDate         DateTime
  endDate           DateTime
  registrationStart DateTime
  registrationEnd   DateTime
  status            SemesterStatus @default(UPCOMING)

  sections    Section[]
  students    StudentProfile[]
  feeInvoices FeeInvoice[]

  createdAt DateTime  @default(now())
  updatedAt DateTime  @updatedAt
  deletedAt DateTime?

  @@index([status])
  @@index([startDate])
  @@map("semesters")
}

model Section {
  id          String        @id @default(uuid())
  courseId    String
  course      Course        @relation(fields: [courseId], references: [id])
  semesterId  String
  semester    Semester      @relation(fields: [semesterId], references: [id])
  sectionName String
  capacity    Int
  room        String?
  schedule    Json?
  status      SectionStatus @default(DRAFT)

  facultyAssignments SectionFaculty[]
  enrollments        Enrollment[]
  attendances        Attendance[]
  exams              Exam[]

  createdAt DateTime  @default(now())
  updatedAt DateTime  @updatedAt
  deletedAt DateTime?

  @@unique([courseId, semesterId, sectionName])
  @@index([courseId])
  @@index([semesterId])
  @@index([status])
  @@map("sections")
}

model SectionFaculty {
  id        String         @id @default(uuid())
  sectionId String
  section   Section        @relation(fields: [sectionId], references: [id])
  facultyId String
  faculty   FacultyProfile @relation(fields: [facultyId], references: [id])
  isPrimary Boolean        @default(true)

  createdAt DateTime @default(now())

  @@unique([sectionId, facultyId])
  @@index([sectionId])
  @@index([facultyId])
  @@map("section_faculty")
}

model Enrollment {
  id         String           @id @default(uuid())
  studentId  String
  student    StudentProfile   @relation(fields: [studentId], references: [id])
  sectionId  String
  section    Section          @relation(fields: [sectionId], references: [id])
  status     EnrollmentStatus @default(ENROLLED)
  enrolledAt DateTime         @default(now())
  droppedAt  DateTime?
  finalGrade Grade?
  gradePoint Float?

  attendances Attendance[]
  results     Result[]

  createdAt DateTime  @default(now())
  updatedAt DateTime  @updatedAt
  deletedAt DateTime?

  @@unique([studentId, sectionId])
  @@index([studentId])
  @@index([sectionId])
  @@index([status])
  @@map("enrollments")
}

// One record per student, per section, per class date
model Attendance {
  id           String           @id @default(uuid())
  enrollmentId String
  enrollment   Enrollment       @relation(fields: [enrollmentId], references: [id])
  studentId    String
  student      StudentProfile   @relation(fields: [studentId], references: [id])
  sectionId    String
  section      Section          @relation(fields: [sectionId], references: [id])
  classDate    DateTime
  status       AttendanceStatus
  markedById   String
  remarks      String?

  createdAt DateTime  @default(now())
  updatedAt DateTime  @updatedAt
  deletedAt DateTime?

  @@unique([studentId, sectionId, classDate])
  @@index([sectionId])
  @@index([studentId])
  @@index([classDate])
  @@index([status])
  @@map("attendance")
}

model Exam {
  id         String     @id @default(uuid())
  sectionId  String
  section    Section    @relation(fields: [sectionId], references: [id])
  examType   ExamType
  title      String?
  examDate   DateTime
  totalMarks Float
  weightage  Float?
  status     ExamStatus @default(DRAFT)

  results Result[]

  createdAt DateTime  @default(now())
  updatedAt DateTime  @updatedAt
  deletedAt DateTime?

  @@index([sectionId])
  @@index([examDate])
  @@index([status])
  @@map("exams")
}

model Result {
  id            String         @id @default(uuid())
  examId        String
  exam          Exam           @relation(fields: [examId], references: [id])
  studentId     String
  student       StudentProfile @relation(fields: [studentId], references: [id])
  enrollmentId  String
  enrollment    Enrollment     @relation(fields: [enrollmentId], references: [id])
  marksObtained Float
  grade         Grade?
  gradePoint    Float?
  status        ResultStatus   @default(DRAFT)
  publishedAt   DateTime?
  enteredById   String

  createdAt DateTime  @default(now())
  updatedAt DateTime  @updatedAt
  deletedAt DateTime?

  @@unique([examId, studentId])
  @@index([examId])
  @@index([studentId])
  @@index([status])
  @@map("results")
}

model FeeInvoice {
  id            String         @id @default(uuid())
  invoiceNumber String         @unique
  studentId     String
  student       StudentProfile @relation(fields: [studentId], references: [id])
  semesterId    String?
  semester      Semester?      @relation(fields: [semesterId], references: [id])
  description   String
  amount        Decimal        @db.Decimal(10, 2)
  dueDate       DateTime
  status        InvoiceStatus  @default(PENDING)

  payments Payment[]

  createdAt DateTime  @default(now())
  updatedAt DateTime  @updatedAt
  deletedAt DateTime?

  @@index([studentId])
  @@index([status])
  @@index([dueDate])
  @@map("fee_invoices")
}

model Payment {
  id               String         @id @default(uuid())
  transactionId    String         @unique
  invoiceId        String
  invoice          FeeInvoice     @relation(fields: [invoiceId], references: [id])
  studentId        String

  amount           Decimal        @db.Decimal(10, 2)
  gateway          PaymentGateway
  status           PaymentStatus  @default(INITIATED)
  gatewayReference String?
  gatewayResponse  Json?
  paidAt           DateTime?

  createdAt DateTime  @default(now())
  updatedAt DateTime  @updatedAt
  deletedAt DateTime?

  @@index([invoiceId])
  @@index([studentId])
  @@index([status])
  @@index([transactionId])
  @@map("payments")
}
